CREATE TABLE Playlists3 (
    id INT PRIMARY KEY,
    user_id INT,
    playlist_name VARCHAR(100),
    total_likes INT
);

SELECT 
    ROW_NUMBER() OVER (ORDER BY total_likes DESC) AS row_num,
    id,
    playlist_name,
    user_id,
    total_likes
FROM Playlists3;

SELECT 
    playlist_name,
    user_id,
    total_likes,
    RANK() OVER (ORDER BY total_likes DESC) AS global_rank
FROM Playlists3;

SELECT 
    playlist_name,
    user_id,
    total_likes,
    DENSE_RANK() OVER (
        PARTITION BY user_id 
        ORDER BY total_likes DESC
    ) AS user_dense_rank
FROM Playlists3;

WITH RankedUserPlaylists AS (
    SELECT 
        user_id,
        playlist_name,
        total_likes,
        ROW_NUMBER() OVER (
            PARTITION BY user_id 
            ORDER BY total_likes DESC, id ASC
        ) AS playlist_rank
    FROM Playlists3
)
SELECT 
    user_id,
    playlist_name,
    total_likes,
    playlist_rank
FROM RankedUserPlaylists
WHERE playlist_rank <= 2
ORDER BY user_id ASC, playlist_rank ASC;
