
CREATE TABLE Restaurant4 (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(150) NOT NULL,
    cuisine VARCHAR(100) NOT NULL,
    location VARCHAR(150) NOT NULL,
    average_rating DECIMAL(3, 2) NOT NULL CHECK (average_rating BETWEEN 0.00 AND 5.00)
);

    COUNT(id) AS restaurant_count
FROM Restaurant4

CREATE TABLE Review2 (
    id INT PRIMARY KEY AUTO_INCREMENT,
    restaurant_id INT NOT NULL,
    user_name VARCHAR(100) NOT NULL,
    rating DECIMAL(2, 1) NOT NULL CHECK (rating BETWEEN 1.0 AND 5.0),
    review_date DATE NOT NULL,
    FOREIGN KEY (restaurant_id) REFERENCES Restaurant4(id) ON DELETE CASCADE
);

SELECT 
    r.name AS restaurant_name,
    r.cuisine,
    ROUND(AVG(rev.rating), 2) AS calculated_avg_rating
FROM Restaurant4 r
JOIN Review2 rev 
    ON r.id = rev.restaurant_id
GROUP BY 
    r.id, 
    r.name, 
    r.cuisine
ORDER BY 
    calculated_avg_rating DESC,
    r.name ASC;
    
    
    WITH RestaurantAggregatedRatings AS (
    SELECT 
        r.id,
        r.name AS restaurant_name,
        r.cuisine,
        ROUND(AVG(rev.rating), 2) AS avg_rating
    FROM Restaurant4 r
    JOIN Review2 rev 
        ON r.id = rev.restaurant_id
    GROUP BY 
        r.id, 
        r.name, 
        r.cuisine
)
SELECT 
    restaurant_name,
    cuisine,
    avg_rating,
    DENSE_RANK() OVER (
        PARTITION BY cuisine 
        ORDER BY avg_rating DESC
    ) AS cuisine_rank
FROM RestaurantAggregatedRatings
ORDER BY 
    cuisine ASC, 
    cuisine_rank ASC;
