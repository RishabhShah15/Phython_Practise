
CREATE TABLE tbl_restaurants (
    res_id INT PRIMARY KEY,
    name VARCHAR(100),
    city VARCHAR(100)
);

CREATE TABLE dishes (
    id INT PRIMARY KEY,
    restaurant_id INT,
    dish_name VARCHAR(100),
    price DECIMAL(10, 2),
    FOREIGN KEY (restaurant_id) REFERENCES tbl_restaurants(res_id)
);

SELECT
    d.dish_name,
    d.price,
    r.name AS restaurant_name,
    r.city
FROM dishes d
INNER JOIN tbl_restaurants r
    ON d.restaurant_id = r.res_id;
    
    SELECT
    r.res_id AS restaurant_id,
    r.name AS restaurant_name,
    r.city,
    d.dish_name,
    d.price
FROM tbl_restaurants r
LEFT JOIN dishes d
    ON r.res_id = d.restaurant_id;
    
    
    SELECT
    d.dish_name,
    d.price,
    d.restaurant_id,
    r.name AS restaurant_name,
    r.city
FROM tbl_restaurants r
RIGHT JOIN dishes d
    ON r.res_id = d.restaurant_id;
