CREATE TABLE Restaurants1 (
    restaurant_id INT PRIMARY KEY,
    restaurant_name VARCHAR(100),
    city VARCHAR(50),
    rating DECIMAL(2, 1)
);

SELECT 
    restaurant_id,
    restaurant_name,
    city,
    rating
FROM Restaurants1 r
WHERE rating > (
    SELECT AVG(rating)
    FROM Restaurants1
    WHERE city = r.city
);

CREATE TABLE Users1 (
    user_id INT PRIMARY KEY,
    user_name VARCHAR(100)
);

CREATE TABLE Orders1 (
    order_id INT PRIMARY KEY,
    user_id INT,
    order_amount DECIMAL(8, 2),
    FOREIGN KEY (user_id) REFERENCES Users1(user_id)
);

    u.user_id,

CREATE TABLE Movies1 (
    movie_id INT PRIMARY KEY,
    title VARCHAR(100),
    language VARCHAR(50)
);

CREATE TABLE Reviews (
    review_id INT PRIMARY KEY,
    movie_id INT,
    rating_stars INT,
    FOREIGN KEY (movie_id) REFERENCES Movies1(movie_id)
);

CREATE TABLE Categories1 (
    category_id INT PRIMARY KEY,
    category_name VARCHAR(50)
);

CREATE TABLE Sellers1 (
    seller_id INT PRIMARY KEY,
    seller_name VARCHAR(100)
);

CREATE TABLE Products1 (
    product_id INT PRIMARY KEY,
    seller_id INT,
    category_id INT,
    product_name VARCHAR(100),
    FOREIGN KEY (seller_id) REFERENCES Sellers1(seller_id),
    FOREIGN KEY (category_id) REFERENCES Categories1(category_id)
);

SELECT s.seller_id, s.seller_name
FROM Sellers1 s
WHERE NOT EXISTS (
    SELECT c.category_id
    FROM Categories1 c
    WHERE NOT EXISTS (
        SELECT 1
        FROM Products1 p
        WHERE p.seller_id = s.seller_id
          AND p.category_id = c.category_id
    )
);

SELECT s.seller_id, s.seller_name
FROM Sellers1 s
WHERE (
    SELECT COUNT(DISTINCT p.category_id)
    FROM Products1 p
    WHERE p.seller_id = s.seller_id
) = (
    SELECT COUNT(*) 
    FROM Categories1
);
