SELECT 
    name,
    location,
    CAST(SUBSTRING_INDEX(rate, '/', 1) AS DECIMAL(3, 1)) AS numeric_rating,
    votes,
    `approx_cost(for two people)` AS approx_cost_for_two
FROM 
    zomato
WHERE 
    location LIKE '%Koramangala%'
    AND cuisines LIKE '%North Indian%'
    AND rate NOT IN ('NEW', '-', '') 
    AND rate IS NOT NULL
ORDER BY 
    numeric_rating DESC,
    votes DESC
LIMIT 5;



WITH RECURSIVE split_cuisines AS (
    SELECT 
        TRIM(SUBSTRING_INDEX(cuisines, ',', 1)) AS cuisine,
        SUBSTRING(cuisines, LENGTH(SUBSTRING_INDEX(cuisines, ',', 1)) + 2) AS remaining_cuisines,
        CAST(REPLACE(`approx_cost(for two people)`, ',', '') AS DECIMAL(10, 2)) AS cost
    FROM zomato
    WHERE `approx_cost(for two people)` IS NOT NULL 
      AND `approx_cost(for two people)` != ''
      AND cuisines IS NOT NULL
    
    UNION ALL
    
    SELECT 
        TRIM(SUBSTRING_INDEX(remaining_cuisines, ',', 1)) AS cuisine,
        CASE 
            WHEN INSTR(remaining_cuisines, ',') > 0 
            THEN SUBSTRING(remaining_cuisines, INSTR(remaining_cuisines, ',') + 1)
            ELSE ''
        END AS remaining_cuisines,
        cost
    FROM split_cuisines
    WHERE remaining_cuisines != ''
)
SELECT 
    cuisine,
    ROUND(AVG(cost), 2) AS avg_cost_for_two,
    COUNT(*) AS total_listings
FROM 
    split_cuisines
WHERE 
    cuisine != ''
GROUP BY 
    cuisine
ORDER BY 
    avg_cost_for_two DESC
LIMIT 3;



SELECT 
    name,
    location,
    cuisines,
    `approx_cost(for two people)` AS cost_for_two,
    CAST(SUBSTRING_INDEX(rate, '/', 1) AS DECIMAL(3, 1)) AS numeric_rating,
    votes
FROM 
    zomato
WHERE 
    online_order = 'Yes'
    AND rate NOT IN ('NEW', '-', '') 
    AND rate IS NOT NULL
    AND CAST(SUBSTRING_INDEX(rate, '/', 1) AS DECIMAL(3, 1)) < 3.0
ORDER BY 
    numeric_rating ASC, 
    votes DESC;
    
    
    
    
    SELECT 
    market_segment,
    COUNT(*) AS restaurant_count
FROM (
    SELECT 
        name,
        CAST(REPLACE(`approx_cost(for two people)`, ',', '') AS UNSIGNED) AS cost_for_two,
        CASE 
            WHEN CAST(REPLACE(`approx_cost(for two people)`, ',', '') AS UNSIGNED) < 400 THEN 'Budget (< 400)'
            WHEN CAST(REPLACE(`approx_cost(for two people)`, ',', '') AS UNSIGNED) BETWEEN 400 AND 800 THEN 'Mid-Range (400 - 800)'
            WHEN CAST(REPLACE(`approx_cost(for two people)`, ',', '') AS UNSIGNED) > 800 THEN 'Premium (> 800)'
        END AS market_segment
    FROM 
        zomato
    WHERE 
        `approx_cost(for two people)` IS NOT NULL 
        AND `approx_cost(for two people)` != ''
) AS segmented_data
GROUP BY 
    market_segment
ORDER BY 
    restaurant_count DESC;
    
    
    SELECT 
    name AS chain_name,
    COUNT(DISTINCT address) AS total_outlets,
    ROUND(AVG(CAST(SUBSTRING_INDEX(rate, '/', 1) AS DECIMAL(3, 1))), 2) AS avg_rating
FROM 
    zomato
WHERE 
    rate NOT IN ('NEW', '-', '') 
    AND rate IS NOT NULL
GROUP BY 
    name
ORDER BY 
    total_outlets DESC
LIMIT 5;
