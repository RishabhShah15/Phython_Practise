CREATE TABLE Users2 (
    user_id INT PRIMARY KEY,
    username VARCHAR(50),
    follower_count INT
);

CREATE TABLE Posts (
    post_id INT PRIMARY KEY,
    user_id INT,
    caption VARCHAR(255),
    like_count INT,
    FOREIGN KEY (user_id) REFERENCES Users2(user_id)

WITH TopUsers AS (
    SELECT 
        'Top User' AS entity_type,
        username AS title_or_name,
        follower_count AS metric_value
    FROM Users2
    ORDER BY follower_count DESC
    LIMIT 3
),
TopPosts AS (
    SELECT 
        'Top Post' AS entity_type,
        caption AS title_or_name,
        like_count AS metric_value
    FROM Posts
    ORDER BY like_count DESC
    LIMIT 3
)
SELECT entity_type, title_or_name, metric_value FROM TopUsers
UNION ALL
SELECT entity_type, title_or_name, metric_value FROM TopPosts;


WITH RECURSIVE BookingCalendar AS (
    
    SELECT 
        CURRENT_DATE() AS booking_date,
        1 AS day_number
    
    UNION ALL
    
    -- Recursive Step: Add 1 day until 7 days are generated
    SELECT 
        DATE_ADD(booking_date, INTERVAL 1 DAY),
        day_number + 1
    FROM BookingCalendar
    WHERE day_number < 7
)
SELECT 
    day_number,
    booking_date,
    DAYNAME(booking_date) AS day_name
FROM BookingCalendar;


SELECT u.user_id, u.username, u.follower_count
FROM Users2 u
WHERE u.user_id IN (
    SELECT inner_u.user_id 
    FROM (
        SELECT user_id, follower_count 
        FROM Users2 
        WHERE follower_count IS NOT NULL
    ) AS inner_u
    WHERE inner_u.follower_count > 1000
)
AND u.user_id NOT IN (
    SELECT user_id 
    FROM Users2 
    WHERE follower_count <= 1000 OR follower_count IS NULL
);


WITH InfluencerUsers AS (
    SELECT 
        user_id,
        username,
        follower_count
    FROM Users2
    WHERE follower_count > 1000
)
SELECT 
    user_id,
    username,
    follower_count
FROM InfluencerUsers
ORDER BY follower_count DESC;
